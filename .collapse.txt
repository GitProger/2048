
    def collapse(a: List[int], dir: int) -> Tuple[bool, List[int], int]:
        print(a, dir)
        if dir == 0:
            return False, a, 0
        moved = False
        score = 0
        n = len(a)

        i, l = 0, 1
        while i < n:
            l = max(l, i + 1)
            while l < n and a[l] == 0:
                l += 1
            if l == n:
                break
            if a[i] == 0:
                moved = True
                a[i] = a[l]
                a[l] = 0
                continue
            elif a[i] == a[l]:
                moved = True
                a[i] *= 2
                score += a[i]
                a[l] = 0
                i += 1
                l += 1
            else:
                i += 1
        if dir > 0:
            z = a.count(0)
            for i in range(n - z - 1, -1, -1):
                a[i + z] = a[i]
                if i < z:
                    a[i] = 0
        return moved, a, score
